-- создание базы данных bank
create database bank;


-- создание таблицы bank
create table if not exists bank (
	id serial primary key,
	name varchar(50)
);


-- заполнение таблицы банков названиями
insert into bank (name) values
	('ЗАО «МТБанк»'),
	('ЗАО «Альфа-Банк»'),
	('ОАО «БНБ-Банк»'),
	('ОАО «Приорбанк»'),
	('ОАО «Белинвестбанк»'),
	('ОАО «АСБ Беларусбанк»');


-- просмотр информации из таблицы банков по айди и названиям
select id, name from bank;


-- создание таблицы продуктов банка
create table if not exists product (
	id serial primary key,
	bank_id integer references bank (id) on delete cascade, -- каскадное удаление сразу (удалится банк из списка, удалятся все его продукты)
	type_of_product varchar(50),
	name_of_product varchar(300),
	interest_rate decimal(6, 3)
);
