-- создание таблицы продуктов банка
create table if not exists product (
	id serial primary key,
	bank_id integer references bank (id) on delete cascade, -- каскадное удаление сразу (удалится банк из списка, удалятся все его продукты)
	type_of_product varchar(50),
	name_of_product varchar(300),
	interest_rate decimal(6, 3)
);


-- добавление колонки со ставкой в грейс-периоде
alter table product add column interest_rate_grace decimal(6, 3);


-- изменение логики типа колонки с грейс-периодом (было 6 символов и максимум 3 после запятой, стало 10 символов, после запятой можно 6 символов)

-- связь клиентов с продуктами многие ко многим
-- каждый клиент может иметь разные продукты в разных банках, но не может иметь два одинаковых продукта в одном банке
create table if not exists m2m_clients_product(
	client_id integer references clients(id) on delete cascade,
	bank_id integer references bank(id) on delete cascade,
	product_id integer references product(id) on delete cascade,
	opened_date date default current_date,
    closed_date date,
    is_active boolean default true,
	unique(client_id, product_id, bank_id)
);

-- выборка клиентов у которых есть ипотека
select clients.patronymic, clients.first_name, clients.last_name, bank.name as name,
	product.name_of_product, m2m_clients_product.opened_date, m2m_clients_product.closed_date
from clients
	join m2m_clients_product on clients.id = m2m_clients_product.client_id
	join product on m2m_clients_product.product_id = product.id
	join bank on m2m_clients_product.bank_id = bank.id
	where product.type_of_product = 'Ипотека' and m2m_clients_product.is_active = false;


-- создание таблицы сотрудников
create table if not exists employee(
	id serial primary key,
	bank_id integer references bank(id) on delete set null, -- если удалили банк, то банк айди - null, но сотрудник останется
	first_name varchar(50),
    last_name varchar(50),
    patronymic varchar(50),
    position varchar(100),
    passport_number varchar(20),
    phone varchar(20)
);
