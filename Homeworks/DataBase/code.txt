select clients.patronymic, clients.first_name, clients.last_name, bank.name as name,
	product.name_of_product, m2m_clients_product.opened_date, m2m_clients_product.closed_date
from clients
	join m2m_clients_product on clients.id = m2m_clients_product.client_id
	join product on m2m_clients_product.product_id = product.id
	join bank on m2m_clients_product.bank_id = bank.id
	where product.type_of_product = 'Ипотека' and m2m_clients_product.is_active = false;

-- таблица связи один ко многим, где 1 сотрудник может обслуживать многих клиентов
create table if not exists one_to_many_employee_clients(
	client_id integer references clients(id) on delete cascade,
	employee_id integer references employee(id) on delete set null,
	bank_id integer references bank(id) on delete cascade,
	branch_id integer references branch(id) on delete cascade,
	primary key (client_id, employee_id)
);

-- просмотр сколько клиентов у каждого сотрудника
select
    employee.first_name as Имя_сотрудника,
    employee.last_name as Фамилия_сотрудника,
	count(one_to_many_employee_clients.client_id) as Количество_клиентов
from employee
left join one_to_many_employee_clients on employee.id = one_to_many_employee_clients.employee_id
group by employee.id, employee.first_name, employee.last_name
order by Количество_клиентов desc;


-- подсчет общего количества сотрудников по филиалам МТБанка
select
    branch.name as Название_филиала,
    count(employee.id) as Количество_сотрудников
from employee
join branch on employee.branch_id = branch.id
where employee.bank_id = 1
group by branch.id, branch.name
order by branch.name;


-- подсчет количества сотрудников у клиента
select
    clients.id as id_клиента,
    clients.first_name as Имя_клиента,
    clients.last_name as Фамилия_клиента,
    clients.patronymic as Отчество_клиента,
    count(one_to_many_employee_clients.employee_id) as Количество_сотрудников
from clients
inner join one_to_many_employee_clients on clients.id = one_to_many_employee_clients.client_id
group by clients.id, clients.first_name, clients.last_name, clients.patronymic;


-- подсчет количества клиентов у сотрудника
select
    employee.id as id_сотрудника,
    employee.first_name as Имя_сотрудника,
    employee.last_name as Фамилия_сотрудника,
    employee.patronymic as Отчество_сотрудника,
    employee.position as Должность_сотрудника,
    count(one_to_many_employee_clients.client_id) as Количество_клиентов
from employee
left join one_to_many_employee_clients on employee.id = one_to_many_employee_clients.employee_id
group by employee.id, employee.first_name, employee.last_name, employee.patronymic, employee.position
order by Количество_клиентов desc;


-- связь один ко многим, где у разных сотрудников может быть 1 начальник
create table if not exists one_to_many_employee_hierarchy(
	id serial primary key,
	chief_id integer references employee(id) on delete set null, --  айди начальника
	subordinate_id integer references employee(id) on delete cascade, -- айди подчиненного
	unique(chief_id, subordinate_id)
);
